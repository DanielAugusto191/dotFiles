cmake_minimum_required(VERSION 3.16.3)
project("TESTE")

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_CXX_FLAGS "-O3")

message(STATUS "Current Source Directory: ${CMAKE_CURRENT_SOURCE_DIR}")
message(STATUS "Current List Directory: ${CMAKE_CURRENT_LIST_DIR}")

find_package(LLVM REQUIRED CONFIG)
find_package(MLIR REQUIRED CONFIG)

list(APPEND CMAKE_MODULE_PATH "${MLIR_CMAKE_DIR}")
list(APPEND CMAKE_MODULE_PATH "${MLIR_LLVM_DIR}")

include(AddLLVM)
include(AddMLIR)


set(CMAKE_C_COMPILER clang)
set(CMAKE_CXX_COMPILER clang++)

include_directories(${LLVM_INCLUDE_DIRS})
include_directories(${MLIR_INCLUDE_DIRS})

link_directories(${LLVM_LIBRARY_DIRS})
add_definitions(${LLVM_DEFINITIONS})


set(LLVM_LINK_COMPONENTS Support)

get_property(dialect_libs GLOBAL PROPERTY MLIR_DIALECT_LIBS)
get_property(conversion_libs GLOBAL PROPERTY MLIR_CONVERSION_LIBS)

add_llvm_executable(teste main.cpp)

target_link_libraries(teste
  PRIVATE
  MyPass
  MLIROptLib
  MLIRRegisterAllPasses
  MLIRRegisterAllDialects
  ${dialect_libs}
  ${conversion_libs}
)

target_include_directories(teste
    PUBLIC
    lib/
)
mlir_check_link_libraries(teste)

add_subdirectory(lib)
